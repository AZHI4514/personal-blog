src/
└── api/
    ├── request.js        # 统一配置 axios 实例（baseURL、拦截器）
    ├── auth.js           # 认证模块（注册、登录、获取当前用户）
    ├── user.js           # 用户管理（管理员：查询所有用户、修改权限）
    ├── post.js           # 帖子模块（发布、分页查询、详情、删除）
    ├── gallery.js        # 画廊模块（查询所有画作、上传画作）
    ├── music.js          # 音乐模块（查询所有音乐、上传音乐）
    ├── link.js           # 链接集模块（查询、新增、编辑）
    ├── visitor.js        # 访客统计模块（记录访客、获取总数）
    ├── clap.js           # 拍手模块（拍手、获取拍手总数）
    ├── announcement.js   # 公告栏模块（获取、发布、删除）
    └── upload.js         # 文件上传模块（上传图片、上传音乐）
